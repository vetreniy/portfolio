using System;

namespace ConsoleApp1;

class Program
{
    static void Main()
    {
        Console.Write("Введите температуру в градусах Цельсия: ");

        double celsius = Convert.ToDouble(Console.ReadLine());

        double fahrenheit = celsius * 9.0 / 5.0 + 32;

        Console.WriteLine($"{fahrenheit}°F");
    }
}
